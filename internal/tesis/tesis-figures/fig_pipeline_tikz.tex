% Esquema del pipeline GNN-HVA (figura TikZ).
% Requiere en el preambulo: \usepackage{tikz}, \usepackage{xcolor},
% \usetikzlibrary{arrows.meta, positioning, shapes.geometric, fit, calc},
% y \input{tesis-figures/palette}.
%
% Fiel a la tesis (Cap. 4, tres fases + entrada):
%   Entrada (gris): Hamiltoniano H(h) sobre una topologia (grafo de espines).
%   Fase 1 (amarillo): datos de referencia por diagonalizacion exacta / DMRG
%                      -> E_0, gap, observables locales.
%   Fase 2 (naranja): VQE con warm-start descendente sobre el HVA
%                      -> parametros optimos theta*(h) (etiquetas).
%   Fase 3 (violeta): la MPNN, entrenada con el grafo de H (entrada) y las
%                      etiquetas theta* (Fase 2), predice theta_pred en una
%                      inferencia; el circuito HVA evalua la energia.
%
% Flujo:
%   - Fila superior: Entrada -> Fase 1 -> Fase 2 -> Fase 3 (secuencial).
%   - El recuadro de entrenamiento de la MPNN (abajo) recibe el grafo de H
%     (desde la Entrada) y las etiquetas theta* (desde la Fase 2); su salida
%     (MPNN entrenada) alimenta la Fase 3.

\begin{tikzpicture}[
    font=\small,
    stage/.style={
        rectangle, rounded corners=3pt, draw=black, line width=1pt,
        fill=#1, minimum width=3.6cm, minimum height=4.3cm, align=center,
        inner sep=6pt,
    },
    gnnbox/.style={
        rectangle, rounded corners=3pt, draw=black, line width=1pt,
        fill=stagePredBg, minimum width=7.2cm, minimum height=3.0cm,
        align=center, inner sep=6pt,
    },
    phaselabel/.style={font=\footnotesize\bfseries, align=center, text=black},
    subtag/.style={font=\scriptsize, text=black!70, align=center},
    outtag/.style={font=\scriptsize\ttfamily, text=black!55, align=center},
    flow/.style={-{Stealth[length=3.2mm]}, line width=1.2pt, black!80},
    trainflow/.style={-{Stealth[length=3mm]}, line width=1.1pt, dashed,
        stagePred!55!black},
    flowlabel/.style={font=\scriptsize\itshape, text=black!75, align=center,
        fill=white, inner sep=2pt},
    icon/.style={line width=0.9pt},
    node distance=1.7cm,
]

% ===== ENTRADA: Definicion del problema (gris) =====
\node[stage=stageProblemBg] (s1) {};
\node[phaselabel] at ([yshift=0.62cm]s1.north) {Entrada};
\node[phaselabel, font=\scriptsize\bfseries] at ([yshift=0.28cm]s1.north)
    {Definici\'on del problema};
\begin{scope}[shift={($(s1.center)+(0,0.9cm)$)}]
    \foreach \x/\y in {-0.75/0.2, -0.05/0.45, 0.7/0.25, -0.45/-0.4, 0.4/-0.45} {
        \node[circle, fill=stageProblem, draw=black, line width=0.5pt,
              minimum size=6pt, inner sep=0pt] at (\x,\y) {};
    }
    \draw[icon, black!70] (-0.75,0.2)--(-0.05,0.45)--(0.7,0.25);
    \draw[icon, black!70] (-0.75,0.2)--(-0.45,-0.4)--(0.4,-0.45)--(0.7,0.25);
    \draw[icon, black!70] (-0.05,0.45)--(-0.45,-0.4);
\end{scope}
\node[subtag] at ($(s1.center)+(0,-0.5cm)$)
    {Hamiltoniano $H(h)$\\sobre una topolog\'ia\\(red de espines)};
\node[outtag] at ($(s1.center)+(0,-1.6cm)$) {$-J\!\sum\! ZZ - h\!\sum\! X$};

% ===== FASE 1: Datos de referencia (amarillo) =====
\node[stage=stageDataBg, right=of s1] (s2) {};
\node[phaselabel] at ([yshift=0.62cm]s2.north) {Fase 1};
\node[phaselabel, font=\scriptsize\bfseries] at ([yshift=0.28cm]s2.north)
    {Datos de referencia};
\begin{scope}[shift={($(s2.center)+(0,0.9cm)$)}]
    \draw[icon, black!70, -{Stealth[length=1.6mm]}] (-0.8,-0.45)--(0.85,-0.45);
    \draw[icon, black!70, -{Stealth[length=1.6mm]}] (-0.8,-0.45)--(-0.8,0.55);
    \node[font=\tiny, text=black!60] at (0.8,-0.62) {$h$};
    \node[font=\tiny, text=black!60] at (-0.97,0.45) {$E_0$};
    \draw[icon, stageData!55!black, line width=1.3pt]
        (-0.75,0.35) .. controls (-0.2,-0.3) and (0.25,-0.38) .. (0.8,-0.18);
\end{scope}
\node[subtag] at ($(s2.center)+(0,-0.5cm)$)
    {Diagonalizaci\'on exacta\\($N\!\le\!14$) o DMRG:\\estado fundamental por $h$};
\node[outtag] at ($(s2.center)+(0,-1.55cm)$)
    {$E_0,\ \mathrm{gap},$\\$\langle X_i\rangle,\ \langle Z_iZ_{i+1}\rangle$};

% ===== FASE 2: Optimizacion VQE (naranja) =====
% Separacion 30% mayor que el resto (1.7cm -> 2.21cm) entre Fase 1 y Fase 2.
\node[stage=stageVQEBg, right=2.45cm of s2] (s3) {};
\node[phaselabel] at ([yshift=0.62cm]s3.north) {Fase 2};
\node[phaselabel, font=\scriptsize\bfseries] at ([yshift=0.28cm]s3.north)
    {VQE con \emph{warm-start}};
% Icono: circuito HVA = R_ZZ (dos qubits) + R_X (un qubit) por capa.
\begin{scope}[shift={($(s3.center)+(0,0.9cm)$)}]
    \draw[icon, black!75] (-1.1,0.25)--(1.1,0.25);
    \draw[icon, black!75] (-1.1,-0.25)--(1.1,-0.25);
    \node[font=\tiny, text=black!55] at (-1.28,0.25) {$q_0$};
    \node[font=\tiny, text=black!55] at (-1.28,-0.25) {$q_1$};
    % Bloque R_ZZ (abarca los dos qubits)
    \node[draw=black, fill=stageVQE!30, minimum width=0.42cm,
          minimum height=0.95cm, inner sep=0pt, font=\tiny]
          at (-0.35,0) {$R_{ZZ}$};
    % Rotaciones R_X de un qubit
    \node[draw=black, fill=stageVQE!45, minimum size=0.32cm, inner sep=0pt,
          font=\tiny] at (0.55,0.25) {$R_X$};
    \node[draw=black, fill=stageVQE!45, minimum size=0.32cm, inner sep=0pt,
          font=\tiny] at (0.55,-0.25) {$R_X$};
    \node[font=\tiny, text=black!55] at (0,-0.62) {HVA de $p$ capas};
\end{scope}
\node[subtag] at ($(s3.center)+(0,-0.55cm)$)
    {Optimizacion iterativa de\\ los \'angulos del circuito};
\node[outtag, font=\tiny] at ($(s3.center)+(0,-1.6cm)$)
    {$\theta^{*}(h)$ (\'angulos HVA)};

% ===== FASE 3: Prediccion y evaluacion (violeta) =====
\node[stage=stagePredBg, right=of s3] (s4) {};
\node[phaselabel] at ([yshift=0.62cm]s4.north) {Fase 3};
\node[phaselabel, font=\scriptsize\bfseries] at ([yshift=0.28cm]s4.north)
    {Predicci\'on y evaluaci\'on};
% Icono: MPNN entrenada -> theta_pred -> circuito HVA evalua la energia.
\begin{scope}[shift={($(s4.center)+(0,1.2cm)$)}]
    \node[draw=black, fill=stagePred!45, rounded corners=2pt,
          minimum width=0.85cm, minimum height=0.55cm, font=\tiny, inner sep=1pt]
          (gnnmini) at (-0.55,0) {MPNN};
    \node[draw=black, fill=stagePred!20, minimum width=0.55cm,
          minimum height=0.55cm, font=\tiny, inner sep=1pt]
          (hvamini) at (0.92,0) {HVA};
    \draw[icon, black!70, -{Stealth[length=1.0mm]}] (gnnmini.east)--(hvamini.west)
        node[midway, above, font=\tiny, text=stagePred!50!black]
        { $ \theta_{\mathrm{pred}} $ };
\end{scope}
\node[subtag] at ($(s4.center)+(0,-0.55cm)$)
    {La MPNN predice $\theta$ en\\una inferencia; el circuito\\HVA eval\'ua la energ\'ia};

% ===== Flechas de flujo (fila superior, secuencial y completa) =====
\draw[flow] (s1) -- (s2) node[flowlabel, midway, above] {$H(h)$};
\draw[flow] (s2) -- (s3) node[flowlabel, midway, above] {estado\\fundamental};
\draw[flow] (s3) -- (s4) node[flowlabel, midway, above] {circuito\\HVA};

% ===== Recuadro: Entrenamiento de la MPNN, debajo y centrado =====
\coordinate (midbelow) at ($(s2.south)!0.5!(s3.south)$);
\node[gnnbox, anchor=north] (gnn) at ($(midbelow)+(0,-3.1cm)$) {};
\node[phaselabel] at ([yshift=0.42cm]gnn.north)
    {Entrenamiento de la MPNN (Fase 3)};
% arquitectura MPNN a la izquierda
\begin{scope}[shift={($(gnn.center)+(-2.3cm,0.2cm)$)}]
    \node[circle, draw=black, fill=stagePred!25, minimum size=6pt, inner sep=0pt] (n1) at (-0.5,0.35) {};
    \node[circle, draw=black, fill=stagePred!25, minimum size=6pt, inner sep=0pt] (n2) at (-0.5,-0.35) {};
    \node[circle, draw=black, fill=stagePred!45, minimum size=7pt, inner sep=0pt] (n3) at (0.15,0.0) {};
    \draw[icon, black!70] (n1)--(n3); \draw[icon, black!70] (n2)--(n3);
\end{scope}
\node[font=\tiny, text=black!60, align=center]
    at ($(gnn.center)+(-2.15cm,-0.7cm)$) {3$\times$ GINConv + pooling};
% objetivo de entrenamiento, a la derecha
\node[align=center, font=\scriptsize, text=black!75]
    at ($(gnn.center)+(1.2cm,0.25cm)$)
    {aprende $h \mapsto \theta$\\minimizando MSE:\\$\theta_{\mathrm{pred}} \approx \theta^{*}$};

% ===== Flechas de datos de entrenamiento =====
% Entrada del grafo de H (desde la Entrada s1) -> lado izquierdo del recuadro.
\draw[trainflow] (s1.south) |- ($(gnn.west)+(0,0.15cm)$)
    node[flowlabel, pos=0.75] {grafo de $H$};
% Etiquetas theta* (desde la Fase 2 s3) -> cara superior del recuadro, entrando
% en su tercio derecho (el titulo esta centrado) y con la etiqueta sobre el
% tramo vertical de bajada para no solaparse con el titulo.
% Recta vertical de una seccion a la otra: baja directo desde la cara inferior de
% la Fase 2 hasta el borde superior del recuadro, en la misma vertical.
\draw[trainflow] (s3.south) -- (s3.south |- gnn.north)
    node[flowlabel, pos=0.55] {$\theta^{*}$ (etiquetas)};

% ===== MPNN entrenada -> Fase 3 (predice los angulos) =====
\draw[flow, stagePred!55!black]
    (gnn.east) -| (s4.south)
    node[flowlabel, pos=0.30] {MPNN entrenada};

\end{tikzpicture}
